rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function to check if user is admin or system_admin
    function isAdmin() {
      return request.auth != null && 
             get(/databases/$(database)/documents/inspectors/$(request.auth.uid)).data.role in ['admin', 'system_admin'];
    }
    
    function isSystemAdmin() {
      return request.auth != null && 
             get(/databases/$(database)/documents/inspectors/$(request.auth.uid)).data.role == 'system_admin';
    }
    
    // PUBLIC ACCESS - For status page
    match /trails/{document} {
      allow read: if true;
      allow update: if request.auth != null;
      allow create, delete: if isAdmin();
    }
    
    match /shelters/{document} {
      allow read: if true;
      allow update: if request.auth != null;
      allow create, delete: if isAdmin();
    }
    
    match /inspections/{document} {
      allow read: if true;
      allow create, update: if request.auth != null;
      allow delete: if isAdmin();
    }
    
    match /trail_inspections/{document} {
      allow read: if true;
      allow create, update: if request.auth != null;
      allow delete: if isAdmin();
    }
    
    match /shelter_inspections/{document} {
      allow read: if true;
      allow create, update: if request.auth != null;
      allow delete: if isAdmin();
    }
    
    // PROTECTED ACCESS
    match /inspectors/{userId} {
      allow read: if request.auth != null;

      // Only admins create accounts; only a system_admin can grant system_admin
      allow create: if isAdmin() &&
                    (request.resource.data.role != 'system_admin' || isSystemAdmin());

      // Admins can edit anyone (but not newly grant system_admin).
      // Users can only edit their own non-privileged fields (see user-profile.html, theme.js).
      allow update: if (isAdmin() &&
                        (request.resource.data.role != 'system_admin' ||
                         resource.data.role == 'system_admin' ||
                         isSystemAdmin())) ||
                       (request.auth != null && request.auth.uid == userId &&
                        request.resource.data.diff(resource.data).affectedKeys()
                          .hasOnly(['phone', 'preferences']));

      allow delete: if isSystemAdmin();
    }
    
    match /config/{document} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }
    
    match /locations/{document} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }
    
    // INFRACTIONS
    match /infractions/{infractionId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null && 
          (request.auth.uid == resource.data.patrolId || isAdmin());
      allow delete: if isAdmin();
    }
    
    // SIGNALISATIONS
    match /signalisations/{signalisationId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null &&
          get(/databases/$(database)/documents/inspectors/$(request.auth.uid)).data.allowSignalisation == true;
      allow update: if request.auth != null && 
          (request.auth.uid == resource.data.inspectorId || isAdmin());
      allow delete: if isAdmin();
    }
    
    // REQUESTS
    match /requests/{requestId} {
      allow read: if request.auth != null && 
          (resource.data.createdBy == request.auth.uid || isAdmin());
      allow create: if request.auth != null;
      allow update: if request.auth != null && 
          (resource.data.createdBy == request.auth.uid || isAdmin());
      allow delete: if isSystemAdmin();
      
      match /comments/{commentId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
        allow update, delete: if false;
      }
    }
    
    // NOTIFICATIONS
    match /notifications/{notificationId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }
    
  }
}

service firebase.storage {
  match /b/{bucket}/o {
    match /{allPaths=**} {
      allow read: if true;

      // Images only, max 15 MB
      allow create, update: if request.auth != null &&
                            request.resource.size < 15 * 1024 * 1024 &&
                            request.resource.contentType.matches('image/.*');

      // Admin only: matches record deletion (inspection-history, inspection-admin orphan cleaner)
      allow delete: if request.auth != null &&
                    firestore.get(/databases/(default)/documents/inspectors/$(request.auth.uid)).data.role in ['admin', 'system_admin'];
    }

    // Inspectors can delete files in their own folder: removing a photo while
    // editing an infraction / signalisation report (PhotoService.deleteRemoved)
    match /infractions/{userId}/{file=**} {
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
    match /signalisations/{userId}/{file=**} {
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
  }
}
